CREATE A TRANSACTION REQUEST
curl --request POST "$TREAD_API_URL/v1/transactions" \
  --header "Authorization: Bearer $TREAD_API_KEY" \
  --header "Idempotency-Key: supplier-invoice-4821" \
  --header "Content-Type: application/json" \
  --data '{
    "assetCode": "KES",
    "amountMinor": "250000",
    "clientReference": "supplier-invoice-4821"
  }'
